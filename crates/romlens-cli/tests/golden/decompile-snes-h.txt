/* snes.h: declarations for the pseudo-C Romlens writes.
*
* The code is for reading. It is valid C, but it is not the game's
* source, and it does not rebuild the ROM.
*/
#ifndef ROMLENS_SNES_H
#define ROMLENS_SNES_H

#include <stdint.h>

typedef uint8_t u8;
typedef uint16_t u16;
typedef uint32_t u32;
typedef int8_t s8;
typedef int16_t s16;
typedef int32_t s32;

/* The CPU. A holds all 16 bits of the accumulator; with 8-bit A only
* its low byte changes. Flags are 0 or 1. */
extern u16 A, X, Y, S, D;
extern u8 DBR;
extern u8 N, V, Z, C;

/* Memory by 24-bit address. Define MEM8 and MEM16 before including
* this to run the code over an array. */
#ifndef MEM8
#define MEM8(a) (*(volatile u8 *)(uintptr_t)(a))
#endif
#ifndef MEM16
#define MEM16(a) (*(volatile u16 *)(uintptr_t)(a))
#endif
#define MEM24(a) ((u32)MEM16(a) | (u32)MEM8((a) + 2) << 16)
/* A 24-bit value held in a three-byte array. */
#define LONG(p) ((u32)(p)[0] | (u32)(p)[1] << 8 | (u32)(p)[2] << 16)

/* Instructions with no C equivalent. */
void SEI(void);
void CLI(void);
void SED(void);
void CLD(void);
void native_mode(void);    /* CLC; XCE */
void emulation_mode(void); /* SEC; XCE */
void push8(u8 v);
void push16(u16 v);
u8 pull8(void);
u16 pull16(void);
void PHP(void);
void PLP(void);
void mvn(u8 dst_bank, u8 src_bank); /* copies A + 1 bytes from X to Y */
void mvp(u8 dst_bank, u8 src_bank);
/* Decimal-mode ADC and SBC of `bits` bits; they set C and V. */
u16 bcd_add(u16 a, u16 b, int bits);
u16 bcd_sub(u16 a, u16 b, int bits);
void WAI(void);
void STP(void);
void BRK(u8 n);
void COP(u8 n);
